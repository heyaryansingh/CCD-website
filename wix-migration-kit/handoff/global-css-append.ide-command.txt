/* ===================================================================
   CCD accessibility fixes - appended 2026-07-30
   Fixes all 8 WCAG 2.1 AA contrast failures. Rationale: ACCESSIBILITY.md sec 2
   Verify with: node wix-migration-kit/handoff/contrast-check.js

   APPEND-ONLY on purpose. These rules sit at the end of global.css and
   override the earlier ones by cascade order, so nothing above needs editing
   and nothing can be accidentally deleted. Do not reorder them above the
   originals.
   =================================================================== */

:root {
  /* --ccd-muted is 5.54:1 on white and PASSES. Left alone deliberately.
     A second token is required because no grey clears 4.5:1 against BOTH
     white and #1a1a1a - lightening the shared token would fix the footer
     and break the on-white use (3.85:1). */
  --ccd-muted-on-dark: #77848a;  /* 4.52:1 on #1a1a1a */
  --ccd-green-deep:    #124a34;  /* 10.22:1 on white - ALL green text */
  --ccd-green-band:    #1c865a;  /* 4.56:1 with white - band w/ body text */
  --ccd-blue-link:     #067db0;  /* 4.60:1 on white - was #0797d4 at 3.29:1 */
}

/* 1. Footer copyright bar. Was 3.14:1 - failing live on every page. */
.ccd-footer-muted {
  color: var(--ccd-muted-on-dark);
}

/* 2. Nav hover. Gold underline was 1.57:1 against a 3:1 requirement.
      Weight change is a second cue so hover is not colour-only. */
.ccd-nav-link {
  border-bottom: 2px solid transparent;  /* reserve space, no hover shift */
}
.ccd-nav-link:hover,
.ccd-nav-link:focus-visible,
.ccd-nav-link[aria-current="page"] {
  border-bottom-color: var(--ccd-green-deep);
  font-weight: 600;
}

/* 3. Focus ring. No single colour clears 3:1 on both white and the green
      band, so it follows the same light/dark logic the text already does. */
:where(a, button, input, select, textarea, [tabindex]):focus-visible {
  outline: 2px solid var(--ccd-green-deep);
  outline-offset: 2px;
  border-radius: 2px;
}
.ccd-band-green :where(a, button, input, select, textarea, [tabindex]):focus-visible,
.ccd-footer :where(a, button, input, select, textarea, [tabindex]):focus-visible {
  outline-color: var(--ccd-gold);  /* 6.49:1 on green-deep, 11.05:1 on ink */
}

/* 4. Reduced motion. Does not stop a Wix carousel's own script -
      turn auto-advance off in the Editor as well. */
@media (prefers-reduced-motion: reduce) {
  *, *::before, *::after {
    animation-duration: 0.01ms !important;
    animation-iteration-count: 1 !important;
    transition-duration: 0.01ms !important;
    scroll-behavior: auto !important;
  }
}

/* 5. Tap targets. Matters most for the social icons and utility bar. */
.ccd-btn-gold,
.ccd-nav-link,
.ccd-utility-text {
  min-height: 44px;
  display: inline-flex;
  align-items: center;
}
